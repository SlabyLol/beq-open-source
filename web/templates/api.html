How to call the API

Use your key as a Bearer token (no browser login needed). Replace YOUR-HOST and beq_YOUR_KEY.

cURL

curl -X POST https://YOUR-HOST/api/generate \
  -H "Authorization: Bearer beq_YOUR_KEY" \
  -H "Content-Type: application/json" \
  -d '{"prompt":"Hello Beq","max_tokens":80,"temperature":0.8}'

Python (beq-client package)

pip install -e ./python

from beq_client import Beq

beq = Beq(
    base_url="https://YOUR-HOST",
    api_key="beq_YOUR_KEY",
    language="de",  # en, de, fr, es, it, pt, nl, pl, ru, ja, zh, ko, tr, ar
)

print(beq.generate("Who are you?"))
print(beq.chat("Write a short poem", language="en", max_tokens=100))
print(beq.status())
print(Beq.languages())

HTML + JavaScript (fetch)

Node.js

const res = await fetch("https://YOUR-HOST/api/generate", {
  method: "POST",
  headers: {
    "Authorization": "Bearer beq_YOUR_KEY",
    "Content-Type": "application/json"
  },
  body: JSON.stringify({
    prompt: "Hello Beq",
    max_tokens: 80,
    temperature: 0.8
  })
});
const data = await res.json();
console.log(data.new_text);

Next.js (App Router)

// app/actions.ts
"use server";

export async function askBeq(prompt: string) {
  const res = await fetch("https://YOUR-HOST/api/generate", {
    method: "POST",
    headers: {
      "Authorization": `Bearer ${process.env.BEQ_API_KEY}`,
      "Content-Type": "application/json"
    },
    body: JSON.stringify({ prompt, max_tokens: 80, temperature: 0.8 })
  });
  return res.json();
}

Java

var client = HttpClient.newHttpClient();
var request = HttpRequest.newBuilder()
    .uri(URI.create("https://YOUR-HOST/api/generate"))
    .header("Authorization", "Bearer beq_YOUR_KEY")
    .header("Content-Type", "application/json")
    .POST(HttpRequest.BodyPublishers.ofString("""
        {"prompt":"Hello Beq","max_tokens":80,"temperature":0.8}
        """))
    .build();
var response = client.send(request, HttpResponse.BodyHandlers.ofString());
System.out.println(response.body());

.NET (C#)

using var client = new HttpClient();
client.DefaultRequestHeaders.Add("Authorization", "Bearer beq_YOUR_KEY");
var json = """{"prompt":"Hello Beq","max_tokens":80,"temperature":0.8}""";
var content = new StringContent(json, Encoding.UTF8, "application/json");
var response = await client.PostAsync("https://YOUR-HOST/api/generate", content);
var result = await response.Content.ReadAsStringAsync();
Console.WriteLine(result);
